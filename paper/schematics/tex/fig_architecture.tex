% stack of channel cards: #1 x, #2 y (front card, lower-left), #3 size, #4 cards, #5 texture prefix, #6 colour
\newcommand{\cardstack}[6]{%
  \pgfmathtruncatemacro{\nlast}{#4-1}%
  \foreach \k in {\nlast,...,0}{%
    \pgfmathsetmacro{\ox}{#1+\k*0.085}\pgfmathsetmacro{\oy}{#2+\k*0.075}%
    \fill[black!13] (\ox+0.035,\oy-0.035) rectangle ++(#3,#3);
    \ifnum\k<3
      \node[anchor=south west,inner sep=0] at (\ox,\oy) {\includegraphics[width=#3cm,height=#3cm]{#5_c\k}};
    \else
      \fill[#6!22] (\ox,\oy) rectangle ++(#3,#3);
    \fi
    \draw[#6!85!black,line width=0.55pt] (\ox,\oy) rectangle ++(#3,#3);
  }}
