% layout card: #1 x, #2 y (lower-left), #3 size, #4 texture
\newcommand{\layoutcard}[4]{%
  \fill[black!13] (#1+0.03,#2-0.03) rectangle ++(#3,#3);
  \node[anchor=south west,inner sep=0] at (#1,#2) {\includegraphics[width=#3cm,height=#3cm]{#4}};
  \draw[ink!70,line width=0.4pt] (#1,#2) rectangle ++(#3,#3);}
% trapezoid: #1 x left, #2 y centre, #3 width, #4 left half-height, #5 right half-height, #6 colour, #7 text
\newcommand{\trap}[7]{%
  \shade[left color=#6!35,right color=#6!12,draw=#6!80!black,line width=0.5pt]
    (#1,#2-#4) -- (#1+#3,#2-#5) -- (#1+#3,#2+#5) -- (#1,#2+#4) -- cycle;
  \node[lbl] at (#1+#3/2,#2) {#7};}
% step header: #1 x, #2 y (top left inside the box), #3 number, #4 title
\newcommand{\stephead}[4]{%
  \node[num] at (#1+0.25,#2-0.27) {#3};
  \node[lbl,anchor=west,font=\fontsize{8}{9.1}\selectfont\bfseries] at (#1+0.48,#2-0.27) {#4};}
% network layer bar: #1 x, #2 y centre, #3 half-height, #4 colour, #5 text
\newcommand{\layerbar}[5]{%
  \fill[#4!55!black] (#1+0.04,#2-#3-0.04) rectangle ++(0.42,2*#3);
  \shade[left color=#4!45,right color=#4!15,draw=#4!80!black,line width=0.45pt] (#1,#2-#3) rectangle ++(0.42,2*#3);
  \node[rotate=90,font=\fontsize{8}{9.0}\selectfont\bfseries,text=ink] at (#1+0.21,#2) {#5};}
